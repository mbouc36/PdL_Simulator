void calibrateSensors() {
    const float axOffset = -80.29565636555708;
    const float ayOffset = -206.00292507379436;
    const float azOffset = 366.6506686508801;
    const float axScale = 5.959236983628436e-05;
    const float ayScale = 5.959236983628436e-05;
    const float azScale = 5.959236983628436e-05;
    const float gxOffset = -0.43725;
    const float gyOffset = 0.12695;
    const float gzOffset = -0.8841499999999999;
    const float mxOffset = -3974.803421115604;
    const float myOffset = 2169.595729020588;
    const float mzOffset = 917.7595264855938;
    const float mxScale = 0.00027272887993853997;
    const float myScale = 0.0002516077983557662;
    const float mzScale = 0.0002839602872989643;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
