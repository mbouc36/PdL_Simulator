void calibrateSensors() {
    const float axOffset = -653.1798291284867;
    const float ayOffset = -132.0447518203655;
    const float azOffset = 327.4355581975997;
    const float axScale = 5.927786162386774e-05;
    const float ayScale = 5.927786162386774e-05;
    const float azScale = 5.927786162386774e-05;
    const float gxOffset = -0.36185;
    const float gyOffset = 0.060199999999999976;
    const float gzOffset = -0.49510000000000004;
    const float mxOffset = -3723.4743864618245;
    const float myOffset = 1794.9632846008735;
    const float mzOffset = 862.8531866197836;
    const float mxScale = 0.00023100819520320925;
    const float myScale = 0.0002133764915304888;
    const float mzScale = 0.00024002065122850745;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
