void calibrateSensors() {
    const float axOffset = 42.74433908152241;
    const float ayOffset = -191.14902695138335;
    const float azOffset = 147.21648537363217;
    const float axScale = 5.931399289071433e-05;
    const float ayScale = 5.931399289071433e-05;
    const float azScale = 5.931399289071433e-05;
    const float gxOffset = -0.41025000000000006;
    const float gyOffset = 0.018349999999999998;
    const float gzOffset = -0.6218;
    const float mxOffset = -1367.1797728903716;
    const float myOffset = 595.0579373503201;
    const float mzOffset = 5159.359573593179;
    const float mxScale = 0.00019823079849434132;
    const float myScale = 0.00019916083412389833;
    const float mzScale = 0.0002270767977597349;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
