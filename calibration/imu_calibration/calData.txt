void calibrateSensors() {
    const float axOffset = 186.7832830298703;
    const float ayOffset = -248.79394208203666;
    const float azOffset = 506.27721771703546;
    const float axScale = 5.958813063488134e-05;
    const float ayScale = 5.958813063488134e-05;
    const float azScale = 5.958813063488134e-05;
    const float gxOffset = -0.41025000000000006;
    const float gyOffset = 0.018349999999999998;
    const float gzOffset = -0.6218;
    const float mxOffset = -2878.030860696095;
    const float myOffset = 631.9422567466736;
    const float mzOffset = 3662.529025128255;
    const float mxScale = 0.00020323858269596627;
    const float myScale = 0.00018406624675236478;
    const float mzScale = 0.00022790864856201426;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
