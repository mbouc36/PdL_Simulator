void calibrateSensors() {
    const float axOffset = 0.8723751626603189;
    const float ayOffset = -183.23544730602953;
    const float azOffset = 881.481569007341;
    const float axScale = 5.9659390745965294e-05;
    const float ayScale = 5.9659390745965294e-05;
    const float azScale = 5.9659390745965294e-05;
    const float gxOffset = -0.39549999999999996;
    const float gyOffset = 0.08355;
    const float gzOffset = -0.6143;
    const float mxOffset = -3763.4454508272374;
    const float myOffset = 2212.3937153285174;
    const float mzOffset = 917.3982978118722;
    const float mxScale = 0.00027538345045815025;
    const float myScale = 0.00023497062460401316;
    const float mzScale = 0.0002794274729611832;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
