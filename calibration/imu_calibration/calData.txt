void calibrateSensors() {
    const float axOffset = -0.12749963590992996;
    const float ayOffset = 0.004628261390296262;
    const float azOffset = 0.018810479764152066;
    const float axScale = 0.9717699382532385;
    const float ayScale = 0.9717699382532385;
    const float azScale = 0.9717699382532385;
    const float gxOffset = -0.19715;
    const float gyOffset = 0.08405000000000001;
    const float gzOffset = -0.0357;
    const float mxOffset = 24.715888893530302;
    const float myOffset = -20.618728742104583;
    const float mzOffset = 71.607200533805;
    const float mxScale = 0.05943439482284269;
    const float myScale = 0.03139906184988591;
    const float mzScale = 0.02254692559476462;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
