void calibrateSensors() {
    const float axOffset = 338.8852913140454;
    const float ayOffset = 331.68214839781376;
    const float azOffset = 611.928244400322;
    const float axScale = 5.967707244054786e-05;
    const float ayScale = 5.967707244054786e-05;
    const float azScale = 5.967707244054786e-05;
    const float gxOffset = -0.386;
    const float gyOffset = -0.20335;
    const float gzOffset = 0.50765;
    const float mxOffset = -1434.8728496227704;
    const float myOffset = 339.0035654940707;
    const float mzOffset = 1237.1579270640023;
    const float mxScale = 0.0002807226713972473;
    const float myScale = 0.00024372786296873277;
    const float mzScale = 0.0002805330755583702;

    AxCal = (AxRaw - axOffset) * axScale;
    AyCal = (AyRaw - ayOffset) * ayScale;
    AzCal = (AzRaw - azOffset) * azScale;
    GxCal = GxRaw*180/PI - gxOffset;
    GyCal = GyRaw*180/PI - gyOffset;
    GzCal = GzRaw*180/PI - gzOffset;
    MxCal = (MxRaw - mxOffset) * mxScale;
    MyCal = (MyRaw - myOffset) * myScale;
    MzCal = (MzRaw - mzOffset) * mzScale;
}
